c3rd_rep_tension.1 = {
	# Optimates demand tax exemptions
	type = country_event
	title = "c3rd_rep_tension.1.t"
	desc = "c3rd_rep_tension.1.desc"
	picture = senate_debate
	right_portrait = scope:c3rd_rep_t1_target_char
	trigger = {
		is_ai = no
		is_republic = yes
		has_variable = republican_tension
		var:republican_tension >= 25
		var:republican_tension < 65
		NOR = {
			has_variable = c3rd_rep_t1_cooldown
			has_variable = c3rd_rep_crisis_fired
		}
		OR = {
			any_character = {
				OR = {
					has_any_office = yes
					is_head_of_family = yes
					is_governor = yes
				}
				character_party = {
					OR = {
						is_party_type = oligarchic_party
						is_party_type = optimates_party
					}
				}
				is_ruler = no
				loyalty <= 50
				NOT = {has_variable = c3rd_rep_t1_target}
			}
			any_governorships = {
				has_variable = governorship_owner
				var:governorship_owner = {
					character_party = {
						OR = {
							is_party_type = oligarchic_party
							is_party_type = optimates_party
						}
					}
					is_ruler = no
					loyalty <= 50
					NOT = {has_variable = c3rd_rep_t1_target}
				}
			}
		}
	}
	weight_multiplier = {
		modifier = {
			factor = 1.5
			var:republican_tension >= 35
		}
		modifier = {
			factor = 2
			any_character = {
				has_character_modifier = major_influence_aristocrat
				character_party = {
					OR = {
						is_party_type = oligarchic_party
						is_party_type = optimates_party
					}
				}
				loyalty <= 50
			}
		}
	}
	immediate = {
		random_character = {
			limit = {
				OR = {
					has_any_office = yes
					is_head_of_family = yes
					is_governor = yes
					save_temporary_scope_as = gov_tmp
					root = {
						any_governorships = {
							has_variable = governorship_owner
							var:governorship_owner = scope:gov_tmp
						}
					}
				}
				character_party = {
					OR = {
						is_party_type = oligarchic_party
						is_party_type = optimates_party
					}
				}
				is_ruler = no
				loyalty <= 50
				NOT = {has_variable = c3rd_rep_t1_target}
			}
			weight = {
				modifier = {
					add = power_base
				}
				modifier = {
					factor = 2
					has_character_modifier = major_influence_aristocrat
				}
			}
			save_scope_as = c3rd_rep_t1_target_char
			set_variable = {
				name = c3rd_rep_t1_target
				days = 7300
			}

		}
		set_variable = {
			name = c3rd_rep_t1_cooldown
			days = 1825
		}
	}
	option = {
		name = c3rd_rep_tension.1.a
		custom_tooltip = c3rd_rep_tension.1.a.tt1
		add_treasury = {
			value = -50
			multiply = num_of_provinces
		}
		oligarchic_approval_effect = {
			VALUE = party_approval_medium
		}
		democratic_approval_effect = {
			VALUE = party_disapproval_small
		}
		change_variable = {
			name = republican_tension
			add = -2
		}
		scope:c3rd_rep_t1_target_char = {
			add_loyalty = softened
		}
		custom_tooltip = c3rd_rep_tension_minus_2
	}
	option = {
		name = c3rd_rep_tension.1.b
		custom_tooltip = c3rd_rep_tension.1.b.tt1
		oligarchic_approval_effect = {
			VALUE = party_disapproval_medium
		}
		current_ruler = {
			add_corruption = -5
			add_popularity = 10
		}
		change_variable = {
			name = republican_tension
			add = 5
		}
		custom_tooltip = c3rd_rep_tension_plus_5
	}
	after = {
		if = {
			limit = {
				var:republican_tension > 100
			}
			remove_variable = republican_tension
			set_variable = {
				name = republican_tension
				value = 100
			}
		}
		else_if = {
			limit = {
				var:republican_tension < 0
			}
			remove_variable = republican_tension
			set_variable = {
				name = republican_tension
				value = 0
			}
		}
	}
}
