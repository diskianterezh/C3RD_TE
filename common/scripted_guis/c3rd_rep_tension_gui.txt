c3rd_start_monarch_reform = {
	scope = country

	is_valid = {
		custom_tooltip = {
			text = "has_already_reform"
			NOT = { has_variable = on_going_rep_thingy }
		}
		
		OR = {
			custom_tooltip = {
				text = has_stab_and_pop
				AND = {
					NOT = { has_law = lex_hortensia }
					stability >= 60
					current_ruler = { popularity >= 80 }
					has_senate_approval >= 0.80
				}
			}
			custom_tooltip = {
				text = has_stab_and_pop_hort
				AND = {
					has_law = lex_hortensia
					stability >= 60
					current_ruler = { popularity >= 60 }
					has_senate_approval >= 0.65
				}
			}
			
		}
	}

	effect = {
		custom_tooltip = "c3rd_rep_reform_attempt_tt"
		add_political_influence = -80
		add_stability = -5
		
			random_list = {
				# Failure weight — decreases as tension rises (higher tension = easier reform)
				80 = {
					modifier = { add = 10  var:republican_tension <= 30 }
					modifier = { add = -5  var:republican_tension >= 50 }
					modifier = { add = -5  var:republican_tension >= 60 }
					modifier = { add = -5 var:republican_tension >= 70 }
					modifier = { add = -10  var:republican_tension >= 80 }
					modifier = { add = -10  current_ruler = { has_trait = orator } }
					modifier = { add = -8   current_ruler = { has_trait = prominent } }
					modifier = { add = -5   current_ruler = { has_trait = assertive } }
					modifier = { add = -5   current_ruler = { charisma >= 8 } }
					modifier = { add = -10  current_ruler = { charisma >= 12 } }
					modifier = { add = 10   current_ruler = { has_trait = submissive } }
					modifier = { add = 8    current_ruler = { has_trait = corrupt } }
					# Failure result
					change_variable = { name = republican_tension add = 10 }
					if = { limit = { var:republican_tension > 100 }
						remove_variable = republican_tension
						set_variable = { name = republican_tension value = 100 } }
					add_stability = -10
					set_variable = { name = recent_rep_reform  days = 3650 }
					custom_tooltip = "c3rd_rep_reform_fail_tt"
					trigger_event = { id = c3rd_events.50  days = 1 }
				}
				20 = {
					modifier = { add = 5  var:republican_tension >= 55 }
					modifier = { add = 10  var:republican_tension >= 65 }
					modifier = { add = 10  var:republican_tension >= 75 }
					modifier = { add = 15  var:republican_tension >= 85 }
					modifier = { add = 20  var:republican_tension >= 95 }
					# Success result
					custom_tooltip = "c3rd_rep_reform_success_tt"
					set_variable = { name = on_going_rep_thingy  days = 600 }
					trigger_event = { id = c3rd_events.41  days = 90 }
				}
			}
		

	}

}

c3rd_split_the_empire = {
	scope = country
	is_valid = {
		custom_tooltip = {
			text = size_requirement
			num_of_cities >= 1000
		}
		custom_tooltip = {
			text = time_requirement
			current_date >= 900.1.1
		}
		custom_tooltip = {
			text = split_empire_any_capital
			any_owned_province = {
				dominant_province_culture = root.culture
				governorship = {
					NOT = {
						this = root.capital_scope.governorship
					}
				}
			}
		}
	}

	effect = {
		custom_tooltip = "c3rd_split_empire"
		trigger_event = c3rd_other_events.76
	}

}


c3rd_convocatio_populi = {
	scope = country
	is_valid = {
		custom_tooltip = {
			text = has_lex_hortensia
			has_law = lex_hortensia
		}
		
		political_influence >= 60
		custom_tooltip = {
			text = has_recent_convocatio
			NOT = { has_variable = recent_convocatio_populi }
		}
	}

	effect = {
		custom_tooltip = "convocatio_populi"
		custom_tooltip = convocatio_populi_desc
		add_political_influence = -60
		change_variable = { name = republican_tension add = -5 }
		custom_tooltip = c3rd_rep_tension_minus_5
		if = { 
			limit = { var:republican_tension < 0 }
			remove_variable = republican_tension
			set_variable = { name = republican_tension value = 0 } 
		}
		set_variable = { name = recent_convocatio_populi  days = 3650 }
	}

}
